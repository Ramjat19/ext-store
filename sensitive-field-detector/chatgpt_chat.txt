async function getSensitiveFieldsFromGemini(elements) {
  const prompt = `These are DOM fields:\n${JSON.stringify(elements, null, 2)}\nIdentify which fields are sensitive like email, phone number, passwords. Return only the indexes of sensitive ones.`;

  const res = await fetch("https://generativelanguage.googleapis.com/v1beta/models/gemini-pro:generateContent?key=YOUR_GEMINI_API_KEY", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      contents: [{ parts: [{ text: prompt }] }]
    })
  });

  const result = await res.json();
  const textResponse = result.candidates[0]?.content?.parts[0]?.text || '';
  const indices = textResponse.match(/\d+/g)?.map(Number) || [];

  return indices.map(i => elements[i]);
}

const processedElements = new WeakSet(); // Tracks already scanned elements

function extractMetadataFromElement(el, index) {
  const metadata = {
    index,
    tag: el.tagName,
    type: el.type || '',
    id: el.id || '',
    name: el.name || '',
    placeholder: el.placeholder || '',
    ariaLabel: el.getAttribute('aria-label') || '',
    text: (el.innerText || '').trim().slice(0, 100),
    labelText: ''
  };

  const label = document.querySelector(`label[for="${el.id}"]`);
  if (label) metadata.labelText = label.innerText.trim();

  return metadata;
}

function collectNewElementsAndSend() {
  const all = document.querySelectorAll('input, textarea, select, div, span');
  const newElements = [];

  let index = 0;
  all.forEach(el => {
    if (!processedElements.has(el)) {
      processedElements.add(el);
      const data = extractMetadataFromElement(el, index++);
      newElements.push(data);
    }
  });

  if (newElements.length > 0) {
    chrome.runtime.sendMessage({
      action: 'analyzeFields',
      data: newElements
    });
  }
}

// Initial scan
collectNewElementsAndSend();

  if (message.action === 'logSensitiveFields') {
    console.log("🔐 Sensitive Fields Detected:");

  }
});

// 🔁 Watch for new DOM elements (lazy load, infinite scroll, etc)
const observer = new MutationObserver(() => {
  collectNewElementsAndSend();
